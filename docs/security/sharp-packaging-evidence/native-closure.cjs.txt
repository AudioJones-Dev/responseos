const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto'),{createRequire}=require('node:module');
const root=process.cwd(),folder=path.resolve(process.argv[2]),build=JSON.parse(fs.readFileSync(path.join(folder,'build-summary.json'))),runtime=JSON.parse(fs.readFileSync(path.join(folder,'runtime-summary.json')));
const digest=file=>crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex');
function imports(file){
 const b=fs.readFileSync(file),pe=b.readUInt32LE(60);if(b.toString('ascii',pe,pe+4)!=='PE\0\0')throw Error('Not PE: '+file);
 const count=b.readUInt16LE(pe+6),optional=pe+24,size=b.readUInt16LE(pe+20),plus=b.readUInt16LE(optional)===0x20b,dirs=optional+(plus?112:96),sections=[];
 for(let i=0;i<count;i++){const p=optional+size+i*40;sections.push({virtual:b.readUInt32LE(p+12),span:Math.max(b.readUInt32LE(p+8),b.readUInt32LE(p+16)),offset:b.readUInt32LE(p+20)});}
 const offset=rva=>{const section=sections.find(row=>rva>=row.virtual&&rva<row.virtual+row.span);if(!section)throw Error('Unmapped RVA');return section.offset+rva-section.virtual;};
 const cstr=rva=>{const start=offset(rva),end=b.indexOf(0,start);return b.toString('ascii',start,end);};
 const rva=b.readUInt32LE(dirs+8),delay=b.readUInt32LE(dirs+13*8),names=[];if(rva)for(let p=offset(rva);b.readUInt32LE(p)||b.readUInt32LE(p+12);p+=20)names.push({name:cstr(b.readUInt32LE(p+12)),kind:'import'});
 if(delay)for(let p=offset(delay);b.readUInt32LE(p)||b.readUInt32LE(p+4);p+=32){if(!(b.readUInt32LE(p)&1))throw Error('Legacy delay VA requires review');names.push({name:cstr(b.readUInt32LE(p+4)),kind:'delay-import'});}
 return names;
}
(async()=>{
 const source=path.join(build.checkout,'node_modules/@img/sharp-win32-x64/lib'),copied=path.join(runtime.runtime,'node_modules/@img/sharp-win32-x64/lib'),native=fs.readdirSync(source).filter(name=>/\.(dll|node)$/.test(name)),closure=native.map(name=>{
  const file=path.join(source,name),target=path.join(copied,name);return {file:name,bytes:fs.statSync(file).size,sourceSha256:digest(file),standalonePresent:fs.existsSync(target),standaloneSha256:fs.existsSync(target)?digest(target):null,dependencies:imports(file).map(row=>({...row,kind:native.some(name=>name.toLowerCase()===row.name.toLowerCase())?'shipped-native':row.kind,classification:native.some(name=>name.toLowerCase()===row.name.toLowerCase())?'PACKAGE_NATIVE':/^api-ms-|^ext-ms-/i.test(row.name)?'WINDOWS_API_SET':'OS_LIBRARY_OR_OTHER'}))};
 });

 const trace=await nft.nodeFileTrace([path.join(source,'../index.cjs')],{base:build.checkout,processCwd:build.checkout,mixedModules:true});
 const routeTraces=[];function walk(dir){return fs.readdirSync(dir,{withFileTypes:true}).flatMap(item=>item.isDirectory()?walk(path.join(dir,item.name)):item.isSymbolicLink()?[]:[path.join(dir,item.name)]);}
 for(const file of walk(path.join(build.checkout,'.next/server')).filter(name=>name.endsWith('.nft.json'))){const entries=JSON.parse(fs.readFileSync(file)).files.filter(entry=>entry.includes('sharp-win32')&&entry.endsWith('.dll'));if(entries.length)routeTraces.push({trace:path.relative(build.checkout,file).replaceAll('\\','/'),dllEntries:entries});}
 const summary={sourceCommit:build.sourceCommit,node:process.version,sharp:JSON.parse(fs.readFileSync(path.join(build.checkout,'node_modules/sharp/package.json'))).version,closure,automaticNftFiles:[...trace.fileList].filter(file=>file.includes('sharp-win32')),automaticNftWarnings:[...trace.warnings].map(error=>error.message),explicitDllTraceCount:routeTraces.length,routeTraces,completeShippedNativeMatch:closure.every(row=>row.sourceSha256===row.standaloneSha256),scope:'All shipped native DLL/node files and static/delay PE imports; dynamic LoadLibrary and OS-loader closure not universally proven'};
 fs.writeFileSync(path.join(folder,'native-closure.json'),JSON.stringify(summary,null,2)+'\n');
 console.log(JSON.stringify({...summary,routeTraces:routeTraces.slice(0,2),closure:closure.map(row=>({...row,dependencies:row.dependencies.filter(dep=>dep.classification==='PACKAGE_NATIVE')}))},null,2));
 if(!summary.completeShippedNativeMatch||routeTraces.length===0)process.exitCode=1;
})().catch(error=>{console.error(error);process.exitCode=1;});
